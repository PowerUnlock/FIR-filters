`timescale 1ns/1ps

module fir_pipeline #( parameter integer N = 100 ) // N is the number of taps, by default N = 100

( input wire clk,
  input wire rst,
  input wire vin,
  input wire signed [15:0] xin,
  output reg vout,
  output reg signed [15:0] yout
);

//_____________________________________________________________________

// Helping functions

//______________________________________________________________________

function automatic signed [15:0] sat16(input signed [63:0] x); // Truncates a 64-bit number to 16 bits.
        begin // b1
            if (x > 64'sd32767)       sat16 = 16'sd32767;
            else if (x < - 64'sd32768) sat16 = - 16'sd32768;
            else                      sat16 = x[15:0];
        end // b1
    endfunction

//______________________________________________________________________

    function integer clog2; // Computes ceiling of log_2(x)
        input integer x;
        integer r;
        begin // b2
            r = 0;
            x = x - 1;
            while (x > 0) begin // b3
                x = x >> 1;
                r = r + 1;
            end // b3
            clog2 = r;
        end // b2
    endfunction

//________________________________________________________________________

    function integer next_pow2; // Computes 2^(clog2(x))
        input integer x;
        integer p;
        begin // b4
            p = 1;
            while (p < x) p = p << 1;
            next_pow2 = p;
        end // b4
    endfunction

//___________________________________________________________________________

// Local Parameters

//____________________________________________________________________________


    localparam integer P      = next_pow2(N); 
    localparam integer LEVELS = clog2(P);     
    localparam integer VDEPTH = 3 + LEVELS;
    localparam signed [63:0] HALF_LSB = 64'sd8192; // 2^13
    localparam integer       SHIFT    = 14;

//____________________________________________________________________________

// Coefficients + delay line

//____________________________________________________________________________

 

    reg signed [15:0] xdel [0:N-1];
    reg signed [15:0] h    [0:N-1];

    initial begin
        $readmemh("coef_q214_hex.txt", h);
    end

//_______________________________________________________________________________

// Stage 1: Update the delay line

//_______________________________________________________________________________

integer i;

    always @(posedge clk) begin // b5
        if (rst) begin // b6
            for (i=0; i<N; i=i+1) xdel[i] <= 16'sd0;
        end // b6 
        else if (vin) begin // b7
            for (i=N-1; i>0; i=i-1) xdel[i] <= xdel[i-1];
            xdel[0] <= xin;
        end // b7
    end // b5


//________________________________________________________________________________

// Stage 2: Calculating all the products terms involved in convolution of input signal with the filter coefficients.

//________________________________________________________________________________

    reg signed [31:0] prod_r [0:N-1];

    always @(posedge clk) begin // b8
        if (rst) begin // b9
            for (i=0; i<N; i=i+1) prod_r[i] <= 32'sd0;
        end // b9 
        else begin // b10
            for (i=0; i<N; i=i+1)
                prod_r[i] <= xdel[i] * h[i];  
        end // b10
    end // b8

//________________________________________________________________________________

// Stage 3: Sign extension of product terms and increasing the total number of product terms to the next power of 2 by padding with zeros.

//________________________________________________________________________________

reg signed [63:0] tree0 [0:P-1];

    integer t;
    always @(posedge clk) begin // b11
        if (rst) begin // b12
            for (t=0; t<P; t=t+1) tree0[t] <= 64'sd0;
        end // b12
        else begin // b13
            for (t=0; t<P; t=t+1) begin // b14
                if (t < N)
                    tree0[t] <= {{32{prod_r[t][31]}}, prod_r[t]};
                else
                    tree0[t] <= 64'sd0; 
            end // b14
        end // b13
    end // b11

//_______________________________________________________________________________

// Stage 4 to 10 : (adder tree, 128 to 64 to 32 to 16 to 8 to 4 to 2 to 1)

//________________________________________________________________________________

    genvar L;
    generate
        for (L=0; L<LEVELS; L=L+1) begin: GEN_LEVEL // b15
            localparam integer OUT_SZ = (P >> (L+1));
            reg signed [63:0] treeN [0:OUT_SZ-1];

            integer k;
            always @(posedge clk) begin // b16
                if (rst) begin // b17
                    for (k=0; k<OUT_SZ; k=k+1) treeN[k] <= 64'sd0;
                end // b17
                else begin // b18
                    for (k=0; k<OUT_SZ; k=k+1) begin // b19
                        if (L == 0)
                            treeN[k] <= tree0[2*k] + tree0[2*k+1];
                        else
                            treeN[k] <= GEN_LEVEL[L-1].treeN[2*k] + GEN_LEVEL[L-1].treeN[2*k+1];
                    end // b19
                end // b18
            end // b16
        end // b15
    endgenerate

wire signed [63:0] acc_tree = GEN_LEVEL[LEVELS-1].treeN[0];

//__________________________________________________________________________________

// Shift register for vin which will help us find valid output

//__________________________________________________________________________________

    reg [VDEPTH:0] vpipe;

    always @(posedge clk) begin // b20
        if (rst) begin // b21
            vpipe <= {(VDEPTH + 1){1'b0}};
        end // b21
        else begin // b22
            vpipe <= {vpipe[VDEPTH-1:0], vin};
        end // b22
    end // b20

//___________________________________________________________________________________

// Stage 11: Generating an output (in Q(2,14) format)if it came from a valid input

//___________________________________________________________________________________

    reg signed [63:0] acc_abs;
    reg signed [63:0] acc_scaled;

    always @(posedge clk) begin // b23
        if (rst) begin // b24
            yout <= 16'sd0;
            vout <= 1'b0;
        end // b24
        else begin // b25
            vout <= vpipe[VDEPTH-1];

            if (vpipe[VDEPTH-1]) begin // b26
                if (acc_tree >= 0) begin // b27
                    acc_scaled = (acc_tree + HALF_LSB) >>> SHIFT;
                end // b27
                else begin // b28
                    acc_abs    = -acc_tree;
                    acc_scaled = -((acc_abs + HALF_LSB) >>> SHIFT);
                end // b28
                yout <= sat16(acc_scaled);
            end // 26
        end // b25
    end // b23

endmodule
